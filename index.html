import pandas as pd
import numpy as np

# IMET Baseline Genomic Dataset
data = {
    'Ancestry': ['European', 'East Asian', 'South Asian', 'African', 'Latino', 'Indigenous'],
    'Global_Pop_Pct': [16.0, 21.0, 24.0, 18.0, 9.0, 12.0],
    'Biobank_Pct': [75.0, 8.0, 6.0, 5.0, 4.0, 2.0]
}

df = pd.DataFrame(data)
df['Representation_Ratio'] = (df['Biobank_Pct'] / df['Global_Pop_Pct']).round(2)
df['Equity_Alert'] = np.where(df['Representation_Ratio'] < 0.5, 'SEVERE BLINDSPOT', 'OPTIMAL')

print(df[['Ancestry', 'Representation_Ratio', 'Equity_Alert']])
